services:

  backend:
    image: my-backend
    ports:
      - "8080:8080"

      DB_HOST: mysql

    depends_on:
      - mysql
      - redis

  mysql:
    image: mysql:8
    environment:
      MYSQL_ROOT_PASSWORD: password123
      MYSQL_DATABASE: myapp

  redis:
